using System;
using System.Collections.Generic;
class Program
{
    static void Main()
    {
        List<int> Lista = new List<int> { 1, 2, 3, 4, 5, 6 };
        Lista.RemoveAll(x => x % 2 == 0);
        foreach (int numero in Lista)
        {
            Console.Write(numero + " ");
        }
    }
}
